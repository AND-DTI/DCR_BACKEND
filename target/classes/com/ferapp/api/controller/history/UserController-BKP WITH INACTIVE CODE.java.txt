    @PostMapping(value="/changepassword", produces="application/json")
    @Operation(summary = "Resetar senha.")
    @ApiResponses(value = {
        @ApiResponse(responseCode="200", description="Senha alterada com sucesso!"),
        @ApiResponse(responseCode="401", description="Problemas com autorização!"),
    })
    @ResponseStatus(HttpStatus.OK)    
    public ResponseEntity<String> update(@RequestBody LoginRepass userP) throws ParseException{	
				
        
        List<User> users = userService.listByEmail(userP.email());
        boolean passOK = false;
        User user;

        if(users.isEmpty()){
            return ResponseEntity.status(HttpStatus.FORBIDDEN) 
            .header("Accept", "application/json")
            .body("Usuário não cadastrado no sistema FERG.COM");
        }else{ 
            user = users.get(0);
            passOK = encoder.matches(userP.passCur(), user.getPassword());            
        }

        if(passOK){
            user.setPassword(encoder.encode(userP.passNew()));
            userService.save(user);
        }else{
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED) 
            .header("Accept", "application/json")
            .body("Problemas com autorização!"); //.body(ae.getMessage());            
        }
                      
        return ResponseEntity.status(HttpStatus.OK) 
            .header("Accept", "application/json")
            .body("Senha alterada com sucesso!");

    }

    private String doEnvelopUser(UserSE entidade, int action) {//throws IOException{

        String TagTypeAction="", envelope;
        
        if(action == 1){
            TagTypeAction = "editUser";            
        }else{
            TagTypeAction ="newUser";            
        }
        
        String envelopeNew = 
        "<x:Envelope "                                                  +"\n"+
        "    xmlns:x=\"http://schemas.xmlsoap.org/soap/envelope/\""     +"\n"+
        "    xmlns:urn=\"urn:admin\"> "                                 +"\n"+
        "    <x:Header/> "                                              +"\n"+
        "    <x:Body> "                                                 +"\n"+
        "        <urn:"+TagTypeAction+">"                               +"\n"+
        "            <urn:IDUSER>{USER}</urn:IDUSER>"                   +"\n"+
        "            <urn:NAME>{NAME}</urn:NAME>"                       +"\n"+
        "            <urn:LOGIN>{LOGIN}</urn:LOGIN>"                    +"\n"+
        "            <urn:PASS>{PASS}</urn:PASS>"                       +"\n"+
        "            <urn:EMAIL>{EMAIL}</urn:EMAIL>"    	            +"\n"+
        "            <urn:IDAREA>{AREA}</urn:IDAREA>"                   +"\n"+
        "            <urn:IDFUNC>{FUNCTION}</urn:IDFUNC>"               +"\n"+
        "            <urn:IDACCGROUP>{GROUP}</urn:IDACCGROUP>"          +"\n"+ 
        "            <urn:CDLEADER>{LEADER}</urn:CDLEADER>"             +"\n"+ 
        "        </urn:"+TagTypeAction+">"                              +"\n"+                    
        "    </x:Body> "                                                +"\n"+
        "</x:Envelope>";

        String envelopeEdit = 
        "<x:Envelope "                                                  +"\n"+
        "    xmlns:x=\"http://schemas.xmlsoap.org/soap/envelope/\""     +"\n"+
        "    xmlns:urn=\"urn:admin\"> "                                 +"\n"+
        "    <x:Header/> "                                              +"\n"+
        "    <x:Body> "                                                 +"\n"+
        "        <urn:"+TagTypeAction+">"                               +"\n"+
        "            <urn:IDUSER>{USER}</urn:IDUSER>"                   +"\n"+
        "            <urn:NAME>{NAME}</urn:NAME>"                       +"\n"+
        "            <urn:LOGIN>{LOGIN}</urn:LOGIN>"                    +"\n"+                     
        "            <urn:CDLEADER>{LEADER}</urn:CDLEADER>"             +"\n"+ 
        "        </urn:"+TagTypeAction+">"                              +"\n"+                    
        "    </x:Body> "                                                +"\n"+
        "</x:Envelope>";        


        if(action == 1){
            envelope = envelopeEdit
                .replace("{USER}", String.valueOf(entidade.getIduser()))
                .replace("{NAME}", entidade.getName())
                .replace("{LOGIN}", entidade.getUsername());
                //.replace("{LEADER}", entidade.getIdleader()); lider from Front not implemented yet
        }else{            
            envelope = envelopeNew
                .replace("{USER}", String.valueOf(entidade.getIduser()))
                .replace("{NAME}", entidade.getName())
                .replace("{LOGIN}", entidade.getUsername())
                .replace("{PASS}", entidade.getPassword())
                .replace("{EMAIL}", entidade.getEmail())
                .replace("{AREA}", entidade.getIdarea())
                .replace("{FUNCTION}", "F001") // default basic access user
                .replace("{GROUP}", "GR004")  // default basic access group  
                .replace("{LEADER}", entidade.getIdleader());            
        }
        
                
        return envelope;

    }


    private ApiResponseCST postUser(String envelope, String action) throws IOException{ 

        
        ApiResponseCST apiResponse = null;
        String URL = api_se_baseurl + "/se/ws/adm_ws.php";
        String typeAction, Action = "";
        


        try{

            String soapBody = envelope;
            typeAction = action;
            Action = "urn:admin#"+action;


            HttpClient client = new HttpClient();            
            if(ENV.equals("hda")){ Security.configProxyClient(client); }    

            //Set method
            PostMethod method = new PostMethod(URL);
            
            //Set header:
            method.setRequestHeader("SOAPAction", Action);            
            method.setRequestHeader("Authorization", Security.getAuthenticationHeader("SEsuite"));
            method.setRequestHeader("Content-Type", "text/xml; charset=utf-8"); 
            
            //Set body:
            StringRequestEntity strEntity = new StringRequestEntity(soapBody, "text/xml", "UTF-8");                        
            method.setRequestEntity(strEntity);

            //Call request
            client.executeMethod(method);
            int responseStatus = method.getStatusCode();
            
            //Buld response
            String responseBody = method.getResponseBodyAsString();
            //aux.saveFormatedXML(responseBody, "SoapResponse_User"+"-user-"+".XML", pathICE_PROCESSADO, "UTF-8", true);                        
            JsonNode responseNode = Auxiliar.nodeFromXML(responseBody, "/Body/"+typeAction+"Response");
            apiResponse = new ApiResponseCST(
                responseStatus,
                responseNode.get("Status").asText(), 
                responseNode.get("Code").asInt(),
                responseNode.get("Detail").asText(), 
                ""//responseNode.get("RecordKey").asText()
            );                        


        }catch (Exception e) {            
            //logICE.saveLog("Erro ao tentar replicar o usuário para o SoftExpert (postUser)! ["+e+"]");
        }finally{
            
        } 



        return apiResponse;

    }


    private UserSE getUserSE(String iduser) throws IOException, InterruptedException {
        
        HttpClient client = new HttpClient();        
        if(ENV.equals("hda")){ Security.configProxyClient(client); }    
        PostMethod method = new PostMethod(api_se_baseurl+"/v1/dataset-integration/userarea");
        
        UserSE usuario = null;
        

        try{
                                                         
            //Get SE user:
            String jsonParams = "{ \"IDUSER\":"+iduser+" }"; //can use IDUSER="-"            
            method.setRequestHeader("Authorization", Security.getAuthenticationHeader("SEsuite"));
            method.setRequestHeader("Content-Type", "application/json");            
            StringRequestEntity jsonBody = new StringRequestEntity(jsonParams, "application/json", "UTF-8");                        
            method.setRequestEntity(jsonBody);

            client.executeMethod(method);
            String response = method.getResponseBodyAsString();
            //if (method.getStatusCode() == HttpStatus.SC_OK) {
            if (method.getStatusCode() == HttpStatus.ACCEPTED.value()) {
                if(response.equals("[]")){
                    //logICEproc.saveLog("Nenhum usuário encontrado com a matrícula "+iduser+"!. [getUserSE())]");                
                }else{
                    Gson gson = new Gson();
                    UserSE[] usuarios = gson.fromJson(response, UserSE[].class);
                    usuario = usuarios[0];
                }
            }else{
                //logICE.saveLog("Requisição (getUserSE()) não retornou 200 [retorno: "+method.getStatusCode()+ "; "+"resp: "+response+"]"); 
            }


    
        }catch (Exception e){
            //logICE.saveLog("Erro ao recuperar solicitação com a identificação "+iduser+" do SoftExpert (getUserSE())! ["+e+"]"); 
        }finally {
            method.releaseConnection();
        }
            
        return usuario;
        
    }

        "</x:Envelope>";
        
        envelope = envelope
            .replace("{USER}", String.valueOf(entidade.getUserid()))
            .replace("{NAME}", entidade.getName())
            .replace("{LOGIN}", entidade.getUsername())
            .replace("{PASSWORD}", entidade.getPassword())
            .replace("{EMAIL}", entidade.getEmail())
            .replace("{AREA}", entidade.getIdarea())
            .replace("{FUNCTION}", "F001") //default basic access user
            .replace("{GROUP}", "GR004");  //default basic access group
                
        //aux.saveFile(envelope, entidade.getNroit()+"_SOAP.XML", pathICE_PROCESSADO, "UTF-8");


        return envelope;

    }
